
WITH sampled AS (
    SELECT
        cod_apla_risc_bace,

        CASE
            WHEN TRY_CAST(
                cod_prco_risc_bace AS BIGINT
            ) IS NOT NULL
            THEN 'Numeric'
            ELSE COALESCE(
                cod_prco_risc_bace,
                '<NULL>'
            )
        END AS protocol_category,

        dat_hor_cslt_risc_bace

    FROM
        "db_corp_dadoscadastrais_gerirriscobacen_sor_01"
        ."risco_bacen_scr"

    WHERE anomesdia = '20250101'

      AND MOD(
          FROM_BASE(
              SUBSTR(
                  TO_HEX(
                      MD5(
                          TO_UTF8(
                              CAST(cod_idef_pess AS VARCHAR)
                          )
                      )
                  ),
                  1, 8
              ),
              16
          ),
          1000
      ) = 0
)

SELECT
    cod_apla_risc_bace AS acquisition_channel,
    protocol_category,
    dat_hor_cslt_risc_bace AS consultation_timestamp,
    COUNT(*) AS n_sampled_rows

FROM sampled

GROUP BY
    cod_apla_risc_bace,
    protocol_category,
    dat_hor_cslt_risc_bace

ORDER BY
    acquisition_channel,
    protocol_category,
    n_sampled_rows DESC
