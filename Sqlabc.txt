
WITH parsed AS (
    SELECT
        cod_apla_risc_bace,
        cod_prco_risc_bace,
        dat_hor_cslt_risc_bace,

        COALESCE(
            TRY(DATE_PARSE(
                TRIM(dat_hor_cslt_risc_bace),
                '%Y-%m-%d %H:%i:%s'
            )),
            TRY(DATE_PARSE(
                TRIM(dat_hor_cslt_risc_bace),
                '%d.%m.%Y%H:%i:%s'
            )),
            TRY(DATE_PARSE(
                TRIM(dat_hor_cslt_risc_bace),
                '%d.%m.%Y %H:%i:%s'
            ))
        ) AS consultation_ts

    FROM
        "db_corp_dadoscadastrais_gerirriscobacen_sor_01"
        ."risco_bacen_scr"

    WHERE anomesdia = '20250101'
),

lagged AS (
    SELECT
        cod_apla_risc_bace,
        cod_prco_risc_bace,
        dat_hor_cslt_risc_bace,
        consultation_ts,

        DATE_DIFF(
            'day',
            TIMESTAMP '2025-01-31 00:00:00',
            consultation_ts
        ) AS lag_days

    FROM parsed
)

SELECT
    cod_apla_risc_bace AS acquisition_channel,
    cod_prco_risc_bace AS processing_protocol,

    COUNT(*) AS n_rows,

    COUNT_IF(
        consultation_ts IS NULL
    ) AS n_parse_failures,

    MIN(lag_days) AS min_lag_days,

    APPROX_PERCENTILE(
        lag_days, 0.50
    ) AS median_lag_days,

    APPROX_PERCENTILE(
        lag_days, 0.90
    ) AS p90_lag_days,

    MAX(lag_days) AS max_lag_days

FROM lagged

GROUP BY
    cod_apla_risc_bace,
    cod_prco_risc_bace

ORDER BY
    n_rows DESC
