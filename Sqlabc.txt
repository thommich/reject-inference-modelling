SELECT
    cod_apla_risc_bace,
    CASE
        WHEN cod_prco_risc_bace IN ('RI', 'RM', 'RZ', 'Interno', 'Externo')
            THEN cod_prco_risc_bace
        WHEN TRY_CAST(cod_prco_risc_bace AS BIGINT) IS NOT NULL
            THEN 'NUMERIC_PROTOCOL'
        ELSE 'OTHER_OR_NULL'
    END AS protocol_group,
    COUNT(*) AS n_rows,
    COUNT_IF(
        dat_hor_cslt_risc_bace IS NULL
        OR TRIM(dat_hor_cslt_risc_bace) = ''
    ) AS missing_timestamp,
    MIN(TRY_CAST(dat_hor_cslt_risc_bace AS TIMESTAMP)) AS earliest_consultation,
    MAX(TRY_CAST(dat_hor_cslt_risc_bace AS TIMESTAMP)) AS latest_consultation
FROM db_corp_dadoscadastrais_gerirriscobacen_sor_01.risco_bacen_scr
WHERE anomesdia = '20250101'
GROUP BY 1, 2
ORDER BY n_rows DESC;


SELECT
    SUBSTR(dat_hor_cslt_risc_bace, 1, 7) AS consultation_month,
    COUNT(*) AS n_rows,
    APPROX_DISTINCT(cod_idef_pess) AS approx_people
FROM db_corp_dadoscadastrais_gerirriscobacen_sor_01.risco_bacen_scr
WHERE anomesdia = '20250101'
  AND cod_apla_risc_bace = 'Consulta Online'
  AND cod_prco_risc_bace = 'Externo'
GROUP BY 1
ORDER BY 1;


WITH parsed AS (
    SELECT
        cod_idef_pess,
        COALESCE(
            TRY(DATE_PARSE(
                dat_hor_cslt_risc_bace,
                '%Y-%m-%d %H:%i:%s'
            )),
            TRY(DATE_PARSE(
                dat_hor_cslt_risc_bace,
                '%d.%m.%Y%H:%i:%s'
            )),
            TRY(DATE_PARSE(
                dat_hor_cslt_risc_bace,
                '%d.%m.%Y %H:%i:%s'
            ))
        ) AS consultation_ts
    FROM db_corp_dadoscadastrais_gerirriscobacen_sor_01.risco_bacen_scr
    WHERE anomesdia = '20250101'
      AND cod_apla_risc_bace = 'Consulta Online'
      AND cod_prco_risc_bace = 'Externo'
)
SELECT
    DATE_FORMAT(consultation_ts, '%Y-%m') AS consultation_month,
    COUNT(*) AS n_rows,
    APPROX_DISTINCT(cod_idef_pess) AS approx_people
FROM parsed
GROUP BY 1
ORDER BY 1;



SELECT
    CASE
        WHEN REGEXP_LIKE(
            dat_hor_cslt_risc_bace,
            '^\d{4}-\d{2}-\d{2} '
        ) THEN 'ISO_FORMAT'

        WHEN REGEXP_LIKE(
            dat_hor_cslt_risc_bace,
            '^\d{2}\.\d{2}\.\d{4}'
        ) THEN 'BR_FORMAT'

        WHEN dat_hor_cslt_risc_bace IS NULL
          OR TRIM(dat_hor_cslt_risc_bace) = ''
        THEN 'MISSING'

        ELSE 'OTHER'
    END AS timestamp_format,
    COUNT(*) AS n_rows
FROM db_corp_dadoscadastrais_gerirriscobacen_sor_01.risco_bacen_scr
WHERE anomesdia = '20250101'
  AND cod_apla_risc_bace = 'Consulta Online'
  AND cod_prco_risc_bace = 'Externo'
GROUP BY 1
ORDER BY n_rows DESC;


WITH sampled AS (
    SELECT
        cod_idef_pess,
        dat_hor_cslt_risc_bace
    FROM db_corp_dadoscadastrais_gerirriscobacen_sor_01.risco_bacen_scr
    WHERE anomesdia = '20250101'
      AND cod_apla_risc_bace = 'Consulta Online'
      AND cod_prco_risc_bace = 'Externo'
      AND cod_idef_pess IS NOT NULL
      AND MOD(
          FROM_BASE(
              SUBSTR(
                  TO_HEX(MD5(TO_UTF8(cod_idef_pess))),
                  1, 8
              ),
              16
          ),
          100
      ) = 0
),
per_person AS (
    SELECT
        cod_idef_pess,
        COUNT(*) AS n_rows,
        COUNT(DISTINCT dat_hor_cslt_risc_bace) AS n_consultation_timestamps
    FROM sampled
    GROUP BY cod_idef_pess
)
SELECT
    COUNT(*) AS sampled_people,
    COUNT_IF(n_consultation_timestamps > 1)
        AS people_with_multiple_consultations,
    APPROX_PERCENTILE(n_consultation_timestamps, 0.50)
        AS p50_consultations,
    APPROX_PERCENTILE(n_consultation_timestamps, 0.90)
        AS p90_consultations,
    APPROX_PERCENTILE(n_consultation_timestamps, 0.99)
        AS p99_consultations,
    MAX(n_consultation_timestamps)
        AS max_consultations
FROM per_person;



WITH sampled AS (
    SELECT
        cod_idef_pess,
        dat_hor_cslt_risc_bace
    FROM db_corp_dadoscadastrais_gerirriscobacen_sor_01.risco_bacen_scr
    WHERE anomesdia = '20250101'
      AND cod_apla_risc_bace = 'Consulta Online'
      AND cod_prco_risc_bace = 'Externo'
      AND cod_idef_pess IS NOT NULL
      AND MOD(
          FROM_BASE(
              SUBSTR(
                  TO_HEX(MD5(TO_UTF8(cod_idef_pess))),
                  1, 8
              ),
              16
          ),
          100
      ) = 0
),
per_snapshot AS (
    SELECT
        cod_idef_pess,
        dat_hor_cslt_risc_bace,
        COUNT(*) AS n_rows
    FROM sampled
    GROUP BY 1, 2
)
SELECT
    COUNT(*) AS n_snapshots,
    APPROX_PERCENTILE(n_rows, 0.50) AS p50_rows,
    APPROX_PERCENTILE(n_rows, 0.90) AS p90_rows,
    APPROX_PERCENTILE(n_rows, 0.99) AS p99_rows,
    MAX(n_rows) AS max_rows
FROM per_snapshot;



WITH sampled AS (
    SELECT
        cod_idef_pess,
        dat_hor_cslt_risc_bace,
        cod_moda_cred_risc_bace,
        cod_orig_risc_bace,
        cod_inst_finn_risc_bace
    FROM db_corp_dadoscadastrais_gerirriscobacen_sor_01.risco_bacen_scr
    WHERE anomesdia = '20250101'
      AND cod_apla_risc_bace = 'Consulta Online'
      AND cod_prco_risc_bace = 'Externo'
      AND cod_idef_pess IS NOT NULL
      AND MOD(
          FROM_BASE(
              SUBSTR(
                  TO_HEX(MD5(TO_UTF8(cod_idef_pess))),
                  1, 8
              ),
              16
          ),
          100
      ) = 0
),
per_snapshot AS (
    SELECT
        cod_idef_pess,
        dat_hor_cslt_risc_bace,
        COUNT(*) AS n_rows,
        COUNT(DISTINCT cod_moda_cred_risc_bace) AS n_modalities,
        COUNT(DISTINCT cod_orig_risc_bace) AS n_origins,
        COUNT(DISTINCT cod_inst_finn_risc_bace) AS n_institutions,
        COUNT(DISTINCT CONCAT(
            COALESCE(cod_moda_cred_risc_bace, '<NULL>'), '|',
            COALESCE(cod_orig_risc_bace, '<NULL>'), '|',
            COALESCE(cod_inst_finn_risc_bace, '<NULL>')
        )) AS n_distinct_combinations
    FROM sampled
    GROUP BY 1, 2
)
SELECT
    COUNT(*) AS n_snapshots,
    APPROX_PERCENTILE(n_rows, 0.50) AS p50_rows,
    APPROX_PERCENTILE(n_modalities, 0.50) AS p50_modalities,
    APPROX_PERCENTILE(n_origins, 0.50) AS p50_origins,
    APPROX_PERCENTILE(n_institutions, 0.50) AS p50_institutions,
    COUNT_IF(n_rows > n_distinct_combinations)
        AS snapshots_with_repeated_combinations,
    MAX(n_rows) AS max_rows
FROM per_snapshot;



WITH sampled AS (
    SELECT
        cod_idef_pess,
        cod_prco_risc_bace,
        dat_hor_cslt_risc_bace,
        cod_moda_cred_risc_bace,
        cod_orig_risc_bace,
        cod_inst_finn_risc_bace
    FROM db_corp_dadoscadastrais_gerirriscobacen_sor_01.risco_bacen_scr
    WHERE anomesdia = '20250101'
      AND cod_apla_risc_bace = 'Consulta Lote'
      AND TRY_CAST(cod_prco_risc_bace AS BIGINT) IS NOT NULL
      AND cod_idef_pess IS NOT NULL
      AND MOD(
          FROM_BASE(
              SUBSTR(
                  TO_HEX(MD5(TO_UTF8(cod_idef_pess))),
                  1, 8
              ),
              16
          ),
          1000
      ) = 0
),
per_snapshot AS (
    SELECT
        cod_idef_pess,
        cod_prco_risc_bace,
        dat_hor_cslt_risc_bace,
        COUNT(*) AS n_rows,
        COUNT(DISTINCT ROW(
            cod_moda_cred_risc_bace,
            cod_orig_risc_bace,
            cod_inst_finn_risc_bace
        )) AS n_distinct_combinations
    FROM sampled
    GROUP BY 1, 2, 3
)
SELECT
    COUNT(*) AS n_snapshots,
    APPROX_PERCENTILE(n_rows, 0.50) AS p50_rows,
    APPROX_PERCENTILE(n_rows, 0.90) AS p90_rows,
    APPROX_PERCENTILE(n_rows, 0.99) AS p99_rows,
    MAX(n_rows) AS max_rows,
    COUNT_IF(n_rows > n_distinct_combinations)
        AS snapshots_with_repeated_combinations
FROM per_snapshot;
